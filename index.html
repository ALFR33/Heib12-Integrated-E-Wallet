// server.js - السيرفر الحقيقي لمحفظة Heib12 const express = require('express'); const mongoose = require('mongoose'); const bodyParser = require('body-parser'); const cors = require('cors'); const fetch = require('node-fetch'); // لربط بايبال API const app = express(); app.use(express.json()); app.use(cors()); app.use(express.static('public')); // 1. الاتصال بقاعدة البيانات السحابية (MongoDB Atlas) mongoose.connect('mongodb+srv://your_real_db_connection_string', { useNewUrlParser: true, useUnifiedTopology: true }).then(() => { console.log('✅ تم الاتصال بقاعدة بيانات Heib12 السحابية بنجاح'); }).catch(err => { console.log('❌ خطأ في الاتصال بقاعدة البيانات:', err); }); // 2. تصميم مخطط المستخدم (User Schema) const userSchema = new mongoose.Schema({ name: String, email: { type: String, unique: true }, passwordHash: String, balance: { type: Number, default: 0 }, paypalEmail: String, transactions: [{ title: String, amount: Number, type: String, // 'in' or 'out' status: { type: String, default: 'Completed' }, date: { type: Date, default: Date.now } }] }); const User = mongoose.model('User', userSchema); // 3. مسار إيداع الأموال الحقيقي (عبر بوابة دفع مثل PayPal Checkout / Stripe) app.post('/api/deposit', async (req, res) => { const { email, amount } = req.body; try { let user = await User.findOne({ email }); if (!user) return res.status(404).json({ error: 'المستخدم غير موجود' }); user.balance += parseFloat(amount); user.transactions.push({ title: 'إيداع رصيد حقيقي', amount: parseFloat(amount), type: 'in' }); await user.save(); res.json({ success: true, newBalance: user.balance }); } catch (err) { res.status(500).json({ error: 'خطأ في معالجة الإيداع' }); } }); // 4. مسار السحب الحقيقي إلى PayPal (PayPal Payouts API Integration) app.post('/api/withdraw', async (req, res) => { const { email, paypalEmail, amount } = req.body; try { let user = await User.findOne({ email }); if (!user) return res.status(404).json({ error: 'المستخدم غير موجود' }); if (user.balance < amount) { return res.status(400).json({ error: 'رصيدك غير كافٍ لإتمام السحب' }); } // خصم المبلغ من رصيد المستخدم في قاعدة البيانات user.balance -= parseFloat(amount); user.transactions.push({ title: `سحب الأرباح إلى PayPal (${paypalEmail})`, amount: parseFloat(amount), type: 'out', status: 'Pending' }); await user.save(); // ملاحظة: هنا يتم ربط PayPal Payouts REST API مستقبلاً لتحويل الأموال آلياً // const payoutResponse = await sendPayPalPayout(paypalEmail, amount); res.json({ success: true, message: 'تم تقديم طلب السحب ومعالجته عبر بوابة بايبال بنجاح', newBalance: user.balance }); } catch (err) { res.status(500).json({ error: 'خطأ في عملية التحويل' }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`🚀 خادم محفظة Heib12 يعمل بكامل طاقته على المنفذ ${PORT}`); });
